-- RitmoVital: esquema completo para recrear la base de datos en un proyecto externo de Supabase
-- Ejecutar en el editor SQL del proyecto. Incluye tablas, claves foráneas, índices, funciones, triggers y políticas RLS.

CREATE TABLE public.profiles (
  id uuid PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  nombre text NOT NULL DEFAULT '',
  correo text NOT NULL DEFAULT '',
  edad integer NOT NULL DEFAULT 30,
  tipo_sangre text NOT NULL DEFAULT 'O+',
  peso numeric(5,1) NOT NULL DEFAULT 70,
  estatura numeric(3,2) NOT NULL DEFAULT 1.70,
  consentimiento boolean NOT NULL DEFAULT false,
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NOT NULL DEFAULT now()
);
GRANT SELECT, INSERT, UPDATE ON public.profiles TO authenticated;
GRANT ALL ON public.profiles TO service_role;
ALTER TABLE public.profiles ENABLE ROW LEVEL SECURITY;
CREATE POLICY "perfil propio select" ON public.profiles FOR SELECT TO authenticated USING (auth.uid() = id);
CREATE POLICY "perfil propio insert" ON public.profiles FOR INSERT TO authenticated WITH CHECK (auth.uid() = id);
CREATE POLICY "perfil propio update" ON public.profiles FOR UPDATE TO authenticated USING (auth.uid() = id) WITH CHECK (auth.uid() = id);

CREATE TABLE public.mediciones (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id uuid NOT NULL,
  titulo text NOT NULL DEFAULT 'Medición',
  bpm integer NOT NULL,
  contexto text NOT NULL DEFAULT 'Reposo',
  origen text NOT NULL DEFAULT 'manual',
  created_at timestamptz NOT NULL DEFAULT now()
);
CREATE INDEX mediciones_user_idx ON public.mediciones(user_id, created_at DESC);
GRANT SELECT, INSERT, UPDATE, DELETE ON public.mediciones TO authenticated;
GRANT ALL ON public.mediciones TO service_role;
ALTER TABLE public.mediciones ENABLE ROW LEVEL SECURITY;
CREATE POLICY "mediciones propias" ON public.mediciones FOR ALL TO authenticated USING (auth.uid() = user_id) WITH CHECK (auth.uid() = user_id);

CREATE TABLE public.sintomas (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id uuid NOT NULL,
  bpm integer NOT NULL,
  contexto text NOT NULL DEFAULT 'Reposo',
  sintomas text[] NOT NULL DEFAULT ARRAY['Ninguno'],
  nota text NOT NULL DEFAULT '',
  created_at timestamptz NOT NULL DEFAULT now()
);
CREATE INDEX sintomas_user_idx ON public.sintomas(user_id, created_at DESC);
GRANT SELECT, INSERT, UPDATE, DELETE ON public.sintomas TO authenticated;
GRANT ALL ON public.sintomas TO service_role;
ALTER TABLE public.sintomas ENABLE ROW LEVEL SECURITY;
CREATE POLICY "sintomas propios" ON public.sintomas FOR ALL TO authenticated USING (auth.uid() = user_id) WITH CHECK (auth.uid() = user_id);

CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS trigger LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$
BEGIN
  INSERT INTO public.profiles (id, correo, nombre, consentimiento)
  VALUES (NEW.id, COALESCE(NEW.email, ''), COALESCE(NEW.raw_user_meta_data->>'nombre', ''),
          COALESCE((NEW.raw_user_meta_data->>'consentimiento')::boolean, false));
  RETURN NEW;
END; $$;
CREATE TRIGGER on_auth_user_created AFTER INSERT ON auth.users
  FOR EACH ROW EXECUTE FUNCTION public.handle_new_user();

CREATE OR REPLACE FUNCTION public.touch_updated_at()
RETURNS trigger LANGUAGE plpgsql SET search_path = public AS $$
BEGIN NEW.updated_at = now(); RETURN NEW; END; $$;
CREATE TRIGGER profiles_updated BEFORE UPDATE ON public.profiles
  FOR EACH ROW EXECUTE FUNCTION public.touch_updated_at();
